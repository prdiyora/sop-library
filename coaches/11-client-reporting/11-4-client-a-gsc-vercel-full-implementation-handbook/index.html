11.4.G 6. Copy/paste: api/gsc.js

File path exactly api/gsc.js banao:

const SITES = {
  /* team na 13 client projects nu mapping — naam public ma share karvana nahi */
};

export default async function handler(req, res) {
  res.setHeader("Cache-Control", "no-store, max-age=0");
  res.setHeader("Content-Type", "application/json; charset=utf-8");
  if (req.method !== "GET") return res.status(405).json({ ok: false, error: "Method not allowed." });

  const site = String(req.query.site || "Client A");
  const hasStart = Boolean(req.query.start);
  const hasEnd = Boolean(req.query.end);
  // Site-only request = safe health check, full saved history aape.
  // Dashboard hamesha explicit start/end dates pass kare.
  const start = hasStart && hasEnd ? String(req.query.start) : (!hasStart && !hasEnd ? "1900-01-01" : "");
  const end = hasStart && hasEnd ? String(req.query.end) : (!hasStart && !hasEnd ? "9999-12-31" : "");
  const datePattern = /^\d{4}-\d{2}-\d{2}$/;
  if (!SITES[site]) return res.status(400).json({ ok: false, error: "Unknown project selected." });
  if (!datePattern.test(start) || !datePattern.test(end) || start > end) {
    return res.status(400).json({ ok: false, error: "Provide both valid start and end dates, or omit both for a full-history health check." });
  }

  const token = process.env.GITHUB_TOKEN;
  if (!token) return res.status(500).json({ ok: false, error: "GitHub read token is not configured on this Vercel environment." });

  try {
    const url = "https://api.github.com/repos/client-a-store/gsc/contents/data.json?ref=main&nocache=" + Date.now();
    const response = await fetch(url, {
      cache: "no-store",
      headers: {
        Accept: "application/vnd.github+json",
        Authorization: "Bearer " + token,
        "X-GitHub-Api-Version": "2022-11-28",
        "Cache-Control": "no-cache, no-store",
        Pragma: "no-cache"
      }
    });
    if (!response.ok) throw new Error("GitHub snapshot request failed (" + response.status + ").");
    const file = await response.json();
    if (file.encoding !== "base64" || !file.content) throw new Error("GitHub snapshot content is missing or not base64 encoded.");
    const snapshot = JSON.parse(Buffer.from(file.content.replace(/\s/g, ""), "base64").toString("utf8"));
    const project = snapshot.projects && snapshot.projects[site];
    if (!project) throw new Error("No saved GSC snapshot exists for this project yet.");

    const daily = (project.daily || [])
      .filter(row => row.date >= start && row.date <= end)
      .map(row => ({
        date: row.date,
        clicks: Number(row.clicks || 0),
        impressions: Number(row.impressions || 0),
        ctr: Number(row.ctr || 0),
        position: Number(row.position || 0)
      }));

    return res.status(200).json({
      ok: true,
      project: project.name,
      siteUrl: project.siteUrl,
      start,
      end,
      fetchedAt: snapshot.updatedAt,
      daily,
      pages: [],
      source: "github-api-private-snapshot"
    });
  } catch (error) {
    console.error("GSC snapshot load failed:", error);
    return res.status(502).json({ ok: false, error: "Could not load the scheduled GSC snapshot. Check the GitHub token permissions and Vercel Function logs." });
  }
}

11.4.H 7. Copy/paste: api/gsc-health.js

const EXPECTED = ["Client A","client_d","client_d","client_e","client_f","client_g","Client Bbox","client_h","client_i","client_j","client_k","client_l","client_m"];

export default async function handler(req, res) {
  res.setHeader("Cache-Control", "no-store, max-age=0");
  res.setHeader("Content-Type", "application/json; charset=utf-8");
  if (req.method !== "GET") return res.status(405).json({ ok: false, error: "Method not allowed." });
  const token = process.env.GITHUB_TOKEN;
  if (!token) return res.status(500).json({ ok: false, error: "GitHub read token is not configured." });
  try {
    const url = "https://api.github.com/repos/client-a-store/gsc/contents/data.json?ref=main&nocache=" + Date.now();
    const response = await fetch(url, {
      cache: "no-store",
      headers: {
        Accept: "application/vnd.github+json",
        Authorization: "Bearer " + token,
        "X-GitHub-Api-Version": "2022-11-28",
        "Cache-Control": "no-cache, no-store",
        Pragma: "no-cache"
      }
    });
    if (!response.ok) throw new Error("GitHub snapshot request failed (" + response.status + ").");
    const file = await response.json();
    if (file.encoding !== "base64" || !file.content) throw new Error("GitHub snapshot content is missing or not base64 encoded.");
    const snapshot = JSON.parse(Buffer.from(file.content.replace(/\s/g, ""), "base64").toString("utf8"));
    const projects = snapshot.projects || {};
    const checks = EXPECTED.map(key => {
      const project = projects[key];
      const daily = project && Array.isArray(project.daily) ? project.daily : [];
      const latestDate = daily.reduce((latest, row) => row.date > latest ? row.date : latest, "");
      const validRows = daily.every(row =>
        /^\d{4}-\d{2}-\d{2}$/.test(row.date) &&
        Number.isFinite(Number(row.clicks)) &&
        Number.isFinite(Number(row.impressions)) &&
        Number.isFinite(Number(row.ctr)) &&
        Number.isFinite(Number(row.position))
      );
      return { key, present: Boolean(project), rows: daily.length, latestDate, validRows };
    });
    const ok = checks.every(item => item.present && item.validRows);
    return res.status(ok ? 200 : 503).json({
      ok,
      source: "github-api-private-snapshot",
      branch: "main",
      fetchedAt: snapshot.updatedAt || null,
      expectedProjects: EXPECTED.length,
      checkedProjects: checks.length,
      allProjectsValid: ok,
      projects: checks
    });
  } catch (error) {
    console.error("GSC health check failed:", error);
    return res.status(502).json({ ok: false, error: "Could not verify the GitHub GSC snapshot. Check GitHub token permissions and Vercel Function logs." });
  }
}

async function loadGsc(site, start, end) {
  const params = new URLSearchParams({ site, start, end });
  const response = await fetch(`/api/gsc?${params.toString()}`, {
    method: "GET",
    cache: "no-store",
    headers: { "Accept": "application/json" }
  });
  const data = await response.json();
  if (!response.ok || !data.ok) {
    throw new Error(data.error || `GSC API failed (${response.status})`);
  }
  return data;
}

11.4.S 18. Minimum-deployment strategy — badha projects ma reusable

Goal: minimum unnecessary Vercel deployments, GitHub API reads/writes, GSC queries, automation runs ane AI/token usage sathe reliable dashboard.

18.2 Deployment decision rule

  • Data-only change (data.json ma navi rows): Vercel redeploy NA karo. API request vakhte latest GitHub main snapshot vanche.
  • Dashboard/API code change: migration/vercel par ek reviewed commit push karo; Git integration ek Preview deployment banave.
  • Preview pass + release approved: production release process follow karo. Daily data refresh mate deploy NA karo.
  • Same dashboard ma vadhare properties: site config entry umero; darek site mate whole dashboard/API clone NA karo.
  • Alag client/team jena secrets, access rules, billing ke branding alag hoy: isolation joie to alag project/repo vapro.
  • Related code/config edits batch kari ne ek var deploy karo; environment-variable changes ne pan effect mate navi deployment joie.

18.3 GitHub, GSC ane API usage ocho rakho

  1. GSC: daily run ma property dida required date window mate ek Search Analytics query; fakt required dimensions (date) ane metrics. Same run ma identical queries repeat na karo.
  2. Backfill: fakt missing/recent dates ke bounded lookback query karo; date key thi idempotently merge karo.
  3. GitHub: merged JSON memory ma taiyar kari ne badha properties process thaya pachhi ek var commit karo; site/date dida commit nahi.
  4. Scheduler GitHub reads: merge pehla ek var vancho ane commit pachhi validation mate ek var pacho vancho.
  5. Dashboard: fakt selected site ane selected date range fetch karo; portfolio totals na joie to darek page view par badhi 13 sites fetch na karo.
  6. Health: commit pachhi ek var; dar thoda seconds polling nahi.
  7. No-op: content unchanged hoy to commit skip kari ne "no new finalized GSC rows" record karo; no-op mate KYAREY deployment trigger na karo.
  8. Retries: transient 429/5xx ne exponential backoff ane ocha max attempts sathe retry karo; Retry-After honor karo; KYAREY infinite loop nahi.
  9. Existing 05:30 automation duplicate karva second Vercel Cron NA umero.

18.6 Trade-offs ane safeguards

  • Shared multi-site dashboard duplicate deployments ghatare, pan badha sites same release cycle share kare; representative properties test karo.
  • Ek moto JSON simple che ane GitHub operations minimize kare. Response size ane performance measure karya pachhi j project/month pramane partition karo.
  • Serverless instances in-memory cache reliably share na pan kare. Correctness mate memory cache par KYAREY rely na karo.
  • Short caching repeat GitHub reads ghatare pan briefly stale data batavi shake. Freshness matter kare to current no-store behavior j rakho; acceptable hoy to j measured short TTL umero.
  • Current API darek request par private GitHub main/data.json vanche. Safest savings = duplicate browser fetches ocha karva, authorization remove karvo ke failures hide karva nahi.

Reusable principle: one scheduler → one batch fetch → one merge → one commit → one health check; deploy fakt tyare jyare code change thay.
